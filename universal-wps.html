async function pdfAddNumbers(){ await pickPDF(async files=>{ const file=files[0]; showNotif('#️⃣ Adding numbers...'); setProgress(20); try{ const buf=await file.arrayBuffer(); const { PDFDocument, StandardFonts, rgb } = PDFLib; const src=await PDFDocument.load(buf,{ignoreEncryption:true}); const font=await src.embedFont(StandardFonts.HelveticaBold); const pages=src.getPages(); pages.forEach((p, i)=>{ const { width, height } = p.getSize(); const text = `Page ${i+1} of ${pages.length}`; const textWidth = font.widthOfTextAtSize(text, 10); // white background for visibility p.drawRectangle({ x: (width - textWidth)/2 - 6, y: 10, width: textWidth + 12, height: 16, color: rgb(1,1,1), opacity: 0.8, }); p.drawText(text, { x: (width - textWidth)/2, y: 14, size: 10, font, color: rgb(0,0,0) }); }); const bytes=await src.save(); downloadBlob(new Blob([bytes],{type:'application/pdf'}), sanitize(file.name.replace(/\.pdf$/i,''))+'-numbered.pdf'); setProgress(100); showNotif('✅ Page numbers added!'); }catch(e){ showNotif('❌ '+e.message); setProgress(0); } }); }
